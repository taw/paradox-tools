namespace = succession_wars

# A random eligible Christian monarchy assumes the union.
country_event = {
	id = succession_wars.10
	title = succession_wars.10.t
	desc = succession_wars.10.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		is_subject = no
		is_lesser_in_union = no
		NOT = {
			OR = {
				personal_union = 2
				integrated_personal_union = 2
				AND = { personal_union = 1 integrated_personal_union = 1 }
			}
		}
		government = monarchy
		religion_group = christian
		NOT = { has_reform = elective_monarchy }
		NOT = { has_reform = polish_elective_monarchy }
		NOT = { has_government_attribute = has_dutch_election }
		NOT = { is_emperor = yes }
		any_country = {
			is_subject = no
			is_lesser_in_union = no
			same_continent = ROOT
			# Vanilla missions use variable_arithmetic_trigger for scaled development checks.
			variable_arithmetic_trigger = {
				export_to_variable = {
					variable_name = succession_wars_junior_development
					value = total_development
					who = ROOT
				}
				multiply_variable = {
					which = succession_wars_junior_development
					value = 0.5
				}
				export_to_variable = {
					variable_name = succession_wars_senior_development
					value = total_development
				}
				check_variable = {
					which = succession_wars_senior_development
					which = succession_wars_junior_development
				}
			}
			government = monarchy
			religion_group = christian
			# Keep foreign unions geographically and confessionally plausible:
			# any ally may lead one, otherwise the senior must be a same-faith neighbour.
			OR = {
				alliance_with = ROOT
				AND = { religion = ROOT is_neighbor_of = ROOT }
			}
		}
	}
	option = {
		name = succession_wars.10.a
		ai_chance = { factor = 100 }
		# EU4 cannot form a normal nested PU.  Resolve the old senior's AI
		# juniors first, as vanilla does for the Burgundian Inheritance.  Human
		# subjects are intentionally never inherited by an event.
		every_subject_country = {
			limit = {
				ai = yes
				OR = {
					is_subject_of_type = personal_union
					is_subject_of_type = integrated_personal_union
				}
			}
			ROOT = { inherit = PREV }
		}
		random_country = {
			limit = {
			is_subject = no
			is_lesser_in_union = no
			same_continent = ROOT
				variable_arithmetic_trigger = {
					export_to_variable = {
						variable_name = succession_wars_junior_development
						value = total_development
						who = ROOT
					}
					multiply_variable = {
						which = succession_wars_junior_development
						value = 0.5
					}
					export_to_variable = {
						variable_name = succession_wars_senior_development
						value = total_development
					}
					check_variable = {
						which = succession_wars_senior_development
						which = succession_wars_junior_development
					}
			}
			government = monarchy
			religion_group = christian
			OR = {
				alliance_with = ROOT
				AND = { religion = ROOT is_neighbor_of = ROOT }
			}
		}
			save_event_target_as = succession_wars_new_overlord
			create_union = ROOT
			if = {
				# Only report a union that the engine actually created.
				# This is the stock Dynastic-event direction: senior scoped here,
				# original event country in ROOT.
				limit = { senior_union_with = ROOT }
				log = "SuccessionWars: [Root.GetName] became junior partner under [This.GetName] in [GetYear]."
			}
			else = {
				log = "SuccessionWars: FAILED foreign PU for [Root.GetName]; selected senior was [This.GetName] in [GetYear]."
			}
		}
	}
}

# A succession crisis gives rival Christian monarchies a short-lived opening to
# press their own dynastic claims.
country_event = {
	id = succession_wars.11
	title = succession_wars.11.t
	desc = succession_wars.11.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		is_subject = no
		government = monarchy
		religion_group = christian
		NOT = { has_reform = elective_monarchy }
		NOT = { has_reform = polish_elective_monarchy }
		NOT = { has_government_attribute = has_dutch_election }
		any_rival_country = { is_subject = no government = monarchy religion_group = christian }
	}
	option = {
		name = succession_wars.11.a
		ai_chance = { factor = 100 }
		log = "SuccessionWars: Christian rivals claimed the throne of [Root.GetName] in [GetYear]."
		every_rival_country = {
			limit = { is_subject = no government = monarchy religion_group = christian }
			add_casus_belli = { target = ROOT type = cb_claim_throne months = 240 }
		}
	}
}
