namespace = succession_wars

country_event = {
	id = succession_wars.30
	title = succession_wars.30.t
	desc = succession_wars.30.d
	picture = RELIGION_eventPicture
	is_triggered_only = yes
	trigger = {
		is_subject = no
		NOT = { religion_group = christian }
		any_neighbor_country = { is_subject = no religion_group = christian NOT = { alliance_with = ROOT } }
	}
	option = {
		name = succession_wars.30.a
		ai_chance = { factor = 100 }
		log = "SuccessionWars: a crusade was declared against [Root.GetName] in [GetYear]."
		every_neighbor_country = {
			limit = { is_subject = no religion_group = christian NOT = { alliance_with = ROOT } }
			declare_war_with_cb = { who = ROOT casus_belli = cb_succession_wars_holy_war }
		}
	}
}
