country_event = {
	id = succession_wars.91
	title = succession_wars.91.t
	desc = succession_wars.91.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	trigger = {
		is_subject = no
		NOT = { religion_group = christian }
		any_ally = {
			is_subject = no
			variable_arithmetic_trigger = {
				export_to_variable = {
					variable_name = succession_wars_protection_junior_development
					value = total_development
					who = ROOT
				}
				multiply_variable = { which = succession_wars_protection_junior_development value = 4 }
				export_to_variable = { variable_name = succession_wars_protection_senior_development value = total_development }
				check_variable = {
					which = succession_wars_protection_senior_development
					which = succession_wars_protection_junior_development
				}
			}
		}
	}
	option = {
		name = succession_wars.91.a
		ai_chance = { factor = 100 }
		log = "SuccessionWars: [Root.GetName] sought protection from a stronger ally in [GetYear]."
		random_ally = {
			limit = {
				is_subject = no
				variable_arithmetic_trigger = {
					export_to_variable = {
						variable_name = succession_wars_protection_junior_development
						value = total_development
						who = ROOT
					}
					multiply_variable = { which = succession_wars_protection_junior_development value = 4 }
					export_to_variable = { variable_name = succession_wars_protection_senior_development value = total_development }
					check_variable = {
						which = succession_wars_protection_senior_development
						which = succession_wars_protection_junior_development
					}
				}
			}
			vassalize = ROOT
		}
	}
}
