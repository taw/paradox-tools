	id = succession_wars.1
	title = succession_wars.1.t
	desc = succession_wars.1.d
	picture = none
	is_triggered_only = yes
	option = {
		name = succession_wars.1.a
		ai_chance = { factor = 100 }

				trigger = {
					is_subject = no

							export_to_variable = { variable_name = succession_wars_junior_development value = total_development who = ROOT }

							export_to_variable = { variable_name = succession_wars_senior_development value = total_development }
							check_variable = { which = succession_wars_senior_development which = succession_wars_junior_development }
						}

						religion_group = christian
						NOT = { has_reform = elective_monarchy }
						NOT = { has_government_attribute = has_dutch_election }
						NOT = { is_emperor = yes }
					}
				}
				country_event = { id = succession_wars.10 }
			}
			20 = { trigger = { has_estate = estate_nobles } country_event = { id = succession_wars.20 } }
			20 = { trigger = { has_estate = estate_church } country_event = { id = succession_wars.21 } }
			20 = { trigger = { has_estate = estate_burghers } country_event = { id = succession_wars.22 } }

				trigger = { is_subject = no NOT = { religion_group = christian } any_neighbor_country = { is_subject = no religion_group = christian NOT = { alliance_with = ROOT } } }

			}
			100 = {
				trigger = { is_subject = no NOT = { religion_group = muslim } any_neighbor_country = { is_subject = no religion_group = muslim NOT = { alliance_with = ROOT } } }
				country_event = { id = succession_wars.31 }
			}
			100 = {
				trigger = { is_subject = no religion_group = muslim any_neighbor_country = { is_subject = no OR = { religion = hinduism religion = sikhism } NOT = { alliance_with = ROOT } } }
				country_event = { id = succession_wars.32 }
			}
			100 = {
				trigger = { is_subject = no any_neighbor_country = { is_subject = no has_reform = steppe_horde NOT = { alliance_with = ROOT } } }
				country_event = { id = succession_wars.40 }
			}
			35 = { country_event = { id = succession_wars.41 } }
			50 = { trigger = { any_owned_province = { NOT = { religion = ROOT } } } country_event = { id = succession_wars.42 } }
			100 = { trigger = { OR = { is_subject_of_type = vassal is_subject_of_type = march } } country_event = { id = succession_wars.50 } }
			100 = {
				trigger = {
					NOT = { is_emperor = yes }
					any_subject_country = {
						is_subject_other_than_tributary_trigger = yes
						NOT = { is_colonial_nation = yes }
						NOT = { is_subject_of_type = daimyo_vassal }
					}
				}
				country_event = { id = succession_wars.60 }
			}
			40 = { trigger = { any_subject_country = { is_subject_of_type = tributary_state } } country_event = { id = succession_wars.61 } }
			40 = { trigger = { any_subject_country = { is_subject_of_type = daimyo_vassal } } country_event = { id = succession_wars.62 } }
			40 = { trigger = { any_subject_country = { is_colonial_nation = yes } } country_event = { id = succession_wars.70 } }
			40 = { trigger = { num_of_hired_mercenary_companies = 1 } country_event = { id = succession_wars.71 } }
			20 = { trigger = { religion = catholic NOT = { tag = PAP } NOT = { is_excommunicated = yes } } country_event = { id = succession_wars.72 } }
			35 = { country_event = { id = succession_wars.80 } }
			5 = { country_event = { id = succession_wars.90 } }
			75 = {
				trigger = {
					is_subject = no
					NOT = { religion_group = christian }
					any_ally = {
						is_subject = no
						variable_arithmetic_trigger = {
							export_to_variable = { variable_name = succession_wars_protection_junior_development value = total_development who = ROOT }
							multiply_variable = { which = succession_wars_protection_junior_development value = 4 }
							export_to_variable = { variable_name = succession_wars_protection_senior_development value = total_development }
							check_variable = { which = succession_wars_protection_senior_development which = succession_wars_protection_junior_development }
						}
					}
				}
				country_event = { id = succession_wars.91 }
			}
		}
	}

country_event = {

	is_triggered_only = yes
	mean_time_to_happen = { days = 1 }
	option = {
		name = succession_wars.80.a
		ai_chance = { factor = 100 }
		log = "SuccessionWars: [Root.GetName] received a diplomatic reset in [GetYear]."
		# Stock every_ally/break_alliance pattern; PREV is the iterated ally.
		every_ally = { ROOT = { break_alliance = PREV } }
		every_country = {
			remove_rival = ROOT
			add_aggressive_expansion = { who = ROOT value = -25 apply_calc = no }
		}
		every_country = { ROOT = { remove_rival = PREV } }
	}
}

country_event = {
	id = succession_wars.91
	title = succession_wars.91.t
	desc = succession_wars.91.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		is_subject = no
		NOT = { religion_group = christian }
		any_ally = {
			is_subject = no
			variable_arithmetic_trigger = {
				export_to_variable = {
					variable_name = succession_wars_protection_junior_development
					value = total_development
					who = ROOT
				}
				multiply_variable = { which = succession_wars_protection_junior_development value = 4 }
				export_to_variable = { variable_name = succession_wars_protection_senior_development value = total_development }
				check_variable = {
					which = succession_wars_protection_senior_development
					which = succession_wars_protection_junior_development
				}
			}
		}
	}
	option = {
		name = succession_wars.91.a
		ai_chance = { factor = 100 }
		log = "SuccessionWars: [Root.GetName] sought protection from a stronger ally in [GetYear]."
		random_ally = {
			limit = {
				is_subject = no
				variable_arithmetic_trigger = {
					export_to_variable = {
						variable_name = succession_wars_protection_junior_development
						value = total_development
						who = ROOT
					}
					multiply_variable = { which = succession_wars_protection_junior_development value = 4 }
					export_to_variable = { variable_name = succession_wars_protection_senior_development value = total_development }
					check_variable = {
						which = succession_wars_protection_senior_development
						which = succession_wars_protection_junior_development
					}
				}
			}
			vassalize = ROOT
		}
	}
}
